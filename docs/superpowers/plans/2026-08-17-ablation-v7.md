```python
# scripts/v7_run.py
"""Run one arm of the v7 generation for one round.

Usage: python scripts/v7_run.py <arm> <round> [cohort ...]

Every round generates afresh. Rescoring stored answers cannot see a pipeline
change, and the question this generation asks is how much the answer moves
between runs, which only a fresh generation shows. The request cache stays on,
so a prompt that is byte-identical to one already sent replays for free and only
the questions a manipulation actually touches are paid for.
"""
from __future__ import annotations

import json
import sys
import time
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parents[1]))
sys.path.insert(0, str(Path(__file__).resolve().parents[1] / "src"))

from argus import ablation                                    # noqa: E402
from scripts import error_analysis as ea                      # noqa: E402

COHORTS = ("full", "safety", "valuechain")
ROOT = Path("artifacts/ablation/v7")


def out_dir(arm_name: str, cohort: str, round_no: int) -> Path:
    return ROOT / ("%s_%s_r%d" % (arm_name, cohort, round_no)) / arm_name


def run_arm(arm_name: str, round_no: int,
            cohorts: list[str] | None = None) -> dict:
    records = ea._load_records()
    out: dict[str, dict] = {}
    for cohort in (cohorts or list(COHORTS)):
        base = out_dir(arm_name, cohort, round_no)
        ea.RESPONSE_DIR = base / "responses"
        ea.JUDGE_DIR = base / "judge"
        ea.OUT_DIR = base
        ea.RESPONSE_DIR.mkdir(parents=True, exist_ok=True)
        cases = ea.load_cohort(cohort)
        started = time.time()
        with ablation.use(arm_name):
            for index, case in enumerate(cases, 1):
                run, _ = ea.with_retry(ea.run_case, case, force=True)
                if case.get("answer_mode") == "open":
                    ea.with_retry(ea.run_judge, case, run["response"], force=True)
                print("  %-11s r%d %3d/%-3d %-8s llm=%s"
                      % (cohort, round_no, index, len(cases),
                         case["question_id"], run["llm_calls"]), flush=True)
                time.sleep(ea.PACE_S)
        rows = ea.analyse(cases, records)
        report = ea.build_report(cases, rows, cohort=cohort)
        report["arm"] = arm_name
        report["round"] = round_no
        (base / "report.json").write_text(
            json.dumps(report, ensure_ascii=False, indent=1), encoding="utf-8")
        overall = report["crosstab"]["overall"]
        out[cohort] = {"correct": overall["correct"], "n": overall["n"]}
        print("%s r%d %s %d/%d %.0f초" % (arm_name, round_no, cohort,
              overall["correct"], overall["n"], time.time() - started), flush=True)
    return out


def main(argv) -> int:
    arm_name, round_no = argv[0], int(argv[1])
    result = run_arm(arm_name, round_no, argv[2:] or None)
    total = sum(v["correct"] for v in result.values())
    n = sum(v["n"] for v in result.values())
    print("%s r%d 합계 %d/%d" % (arm_name, round_no, total, n))
    return 0


if __name__ == "__main__":
    sys.exit(main(sys.argv[1:]))
```

```python
def _keeps_regenerated(before: list[str], after: list[str]) -> bool:
    """Whether the regenerated answer replaces the original.

    A clean result is always adopted, which is the prior behaviour. Beyond
    that, adoption requires the failing kinds to be a strict subset: a set that
    merely changed is not an improvement, and adopting it would trade one
    defect for another while spending a call to do it.
    See docs/design-rationale.md#regeneration-adoption
    """
    if not after:
        return True
    if not ablation.enabled("regenerate_partial"):
        return False
    return set(after) < set(before)
```

```python
def _compare_pair(picked: list[dict], question: str):
    """The two records a comparison question sets against each other.

    Exactly two records with an amount qualify. With three or more the pair is
    not determined by the question, and naming the wrong two would put a
    confident wrong figure at the head of the evidence, which is worse than
    saying nothing.
    See docs/design-rationale.md#comparison-line
    """
    rows = [r for r in _latest_per_chain(picked) if _amount(r) is not None]
    if len(rows) != 2:
        return None
    return rows[0], rows[1]
```

```python
# src/argus/agent/units.py
"""Restate the scale a periodic table declares in its header. No LLM call.

A periodic report puts the scale in a header above the table and writes the
cell in that scale, so a figure is seven or eight digits and its meaning
depends on a line the model has to carry across the table. A recorded run read
a figure in hundred-million units as if it were in millions and reached the
opposite conclusion on a comparison, with the evidence stating the scale
correctly the whole time.
See docs/design-rationale.md#unit-line
"""
from __future__ import annotations

import re

#: Scales a disclosure table declares, with their value in won. Units outside
#: this table are counts rather than amounts and carry no multiplier.
SCALES = {"천원": 1_000, "백만원": 1_000_000, "억원": 100_000_000,
          "십억원": 1_000_000_000, "조원": 1_000_000_000_000}

_HEADER = re.compile(r"[（(]\s*단위\s*[:：]?\s*([^)）]{1,12}?)\s*[)）]")


def unit_line(evidence: str) -> str:
    """One line naming each scale the evidence declares, in order of first use."""
    seen: list[str] = []
    for match in _HEADER.finditer(evidence or ""):
        name = "".join(match.group(1).split())
        if name in SCALES and name not in seen:
            seen.append(name)
    if not seen:
        return ""
    parts = ["%s = %s원" % (name, format(SCALES[name], ",")) for name in seen]
    return ("- [결정적 단위] 위 표의 금액 단위: " + " · ".join(parts)
            + " (표의 숫자에 이 배수를 곱한 값이 실제 금액이다)")
```

```python
# scripts/v7_rebuild_index.py
"""Rebuild the exchange index so the supply-region field reaches retrieval.

Usage:
    python scripts/v7_rebuild_index.py --check     차이만 보고한다
    python scripts/v7_rebuild_index.py --apply     백업 후 교체한다
    python scripts/v7_rebuild_index.py --restore   백업을 되돌린다

The renderer emits a field the shipped index does not carry, and the repository
contract is that a renderer change is followed by a rebuild rather than by a
change to the test that noticed. Every prior measurement was taken against the
shipped strings, so the swap is reversible and the backup is written first.
"""
from __future__ import annotations

import filecmp
import shutil
import subprocess
import sys
from pathlib import Path

ROOT = Path("artifacts")
PAYLOAD = ("exchange_records.jsonl", "exchange_bm25.json")
STAGE = Path("artifacts/v7_index_stage")


def backup_dir() -> Path:
    return Path("artifacts/v7_index_backup")


def build() -> None:
    STAGE.mkdir(parents=True, exist_ok=True)
    subprocess.run([sys.executable, "scripts/build_index.py",
                    "--group", "exchange", "--out-dir", str(STAGE)], check=True)


def report() -> None:
    for name in PAYLOAD:
        live, fresh = ROOT / name, STAGE / name
        same = live.exists() and fresh.exists() and filecmp.cmp(live, fresh, shallow=False)
        print("  %-26s %s" % (name, "동일" if same else "다름"))


def apply() -> None:
    backup_dir().mkdir(parents=True, exist_ok=True)
    for name in PAYLOAD:
        shutil.copy2(ROOT / name, backup_dir() / name)
        shutil.copy2(STAGE / name, ROOT / name)
    print("교체 완료. 되돌리려면 --restore")


def restore() -> None:
    for name in PAYLOAD:
        shutil.copy2(backup_dir() / name, ROOT / name)
    print("복구 완료")


def main(argv) -> int:
    mode = argv[0] if argv else "--check"
    if mode == "--restore":
        restore()
        return 0
    build()
    report()
    if mode == "--apply":
        apply()
    return 0


if __name__ == "__main__":
    sys.exit(main(sys.argv[1:]))
```

```python
# scripts/v7_report.py
"""Aggregate the v7 grid and apply the adoption rule.

Usage: python scripts/v7_report.py

The rule is in code rather than in a reader's head because it has three parts
and a hand check drops one. A run is a single draw, so the minimum across
rounds is what a single evaluation is expected to produce; a higher mean with
an unchanged minimum buys nothing.
See docs/superpowers/specs/2026-08-17-ablation-v7-design.md
"""
from __future__ import annotations

import io
import json
import sys
from collections import defaultdict
from pathlib import Path

ROOT = Path("artifacts/ablation/v7")
COHORTS = ("full", "safety", "valuechain")
ROUNDS = (1, 2, 3)
#: Width of the judge's own spread, measured twice at three rounds each. An
#: improvement inside it is not distinguishable from judge noise.
JUDGE_BAND = 1


def verdict(arm_rows: dict[str, list[int]], base: list[int],
            stable: set[str], flips: dict[str, set[str]]) -> tuple[bool, list[str]]:
    """Whether an arm is adopted, with the reasons either way."""
    reasons: list[str] = []
    ok = True
    for name, totals in arm_rows.items():
        if not totals or not base:
            return False, ["회차가 모자라 판정할 수 없다"]
        gain = min(totals) - min(base)
        if gain <= 0:
            ok = False
            reasons.append("%s 최솟값이 %d 로 기준 %d 를 넘지 못했다"
                           % (name, min(totals), min(base)))
        elif gain <= JUDGE_BAND:
            ok = False
            reasons.append("%s 최솟값 상승 %d 는 판정기 잡음 폭 %d 안이다"
                           % (name, gain, JUDGE_BAND))
        else:
            reasons.append("%s 최솟값 %d -> %d" % (name, min(base), min(totals)))
        broken = sorted(stable & flips.get(name, set()))
        if broken:
            ok = False
            reasons.append("%s 가 안정 문항을 흔들었다: %s" % (name, " ".join(broken)))
    return ok, reasons


def totals(arm_name: str) -> list[int]:
    out = []
    for round_no in ROUNDS:
        got = 0
        seen = 0
        for cohort in COHORTS:
            path = (ROOT / ("%s_%s_r%d" % (arm_name, cohort, round_no))
                    / arm_name / "report.json")
            if not path.exists():
                break
            overall = json.load(io.open(path, encoding="utf-8"))["crosstab"]["overall"]
            got += overall["correct"]
            seen += overall["n"]
        else:
            out.append(got)
    return out


def wrong_sets(arm_name: str) -> dict[int, set[str]]:
    out: dict[int, set[str]] = {}
    for round_no in ROUNDS:
        bad: set[str] = set()
        for cohort in COHORTS:
            path = (ROOT / ("%s_%s_r%d" % (arm_name, cohort, round_no))
                    / arm_name / "report.json")
            if not path.exists():
                continue
            for row in json.load(io.open(path, encoding="utf-8"))["rows"]:
                if not row.get("correct"):
                    bad.add(row["question_id"])
        if bad or path.exists():
            out[round_no] = bad
    return out


def main(argv) -> int:
    arms = argv or ["A42", "V7-1", "V7-2", "V7-3", "V7-4", "V7-5"]
    base_name = arms[0]
    grid = {name: totals(name) for name in arms}
    print("%-8s %s" % ("arm", "회차별 정답 (112 만점)"))
    for name in arms:
        print("  %-8s %s   최소 %s" % (name, grid[name] or "-",
              min(grid[name]) if grid[name] else "-"))

    flips: dict[str, set[str]] = {}
    counts: dict[str, int] = defaultdict(int)
    for name in arms:
        rounds = wrong_sets(name)
        union = set().union(*rounds.values()) if rounds else set()
        inter = set.intersection(*rounds.values()) if rounds else set()
        flips[name] = union - inter
        for qid in union:
            counts[qid] += 1
    stable = {qid for qid in counts if counts[qid] == 0}

    print("\narm 별 흔들린 문항")
    for name in arms:
        print("  %-8s %d문항  %s" % (name, len(flips[name]),
              " ".join(sorted(flips[name])) or "-"))

    base_stable = set()
    base_rounds = wrong_sets(base_name)
    if base_rounds:
        ever = set().union(*base_rounds.values())
        for cohort in COHORTS:
            path = (ROOT / ("%s_%s_r1" % (base_name, cohort)) / base_name / "report.json")
            if path.exists():
                for row in json.load(io.open(path, encoding="utf-8"))["rows"]:
                    if row["question_id"] not in ever:
                        base_stable.add(row["question_id"])

    print("\n채택 판정 (기준 %s)" % base_name)
    for name in arms[1:]:
        ok, why = verdict({name: grid[name]}, grid[base_name], base_stable, flips)
        print("  %-8s %s" % (name, "채택" if ok else "기각"))
        for line in why:
            print("      %s" % line)
    return 0


if __name__ == "__main__":
    sys.exit(main(sys.argv[1:]))
```
