import json
import pathlib

import pytest

from argus.agent import router

def test_no_false_positive_on_any_evalset_question():
    """🔴 골든이 실답을 요구하는 문항을 하나라도 잡으면 이 arm 은 기각이다."""

```python
# src/argus/agent/aggregate.py
"""집계를 **우리가** 계산해 근거에 한 줄로 싣는다. LLM 0회 · 검색 0회.

v4 의 단일 발견이 *"모델이 숫자를 손에 쥐는 순간 틀린다"* 였다. `A20`(치환)이 이긴
이유는 재계산값을 근거에서 우리가 직접 읽어 **모델의 타이핑을 안 거쳤기** 때문이다.
집계도 같다 — R0(2026-08-16)이 오답 6건에서 항이 근거에 **다 있는데** 합·차·몫만
틀린 것을 실측했고, 우리가 계산하면 여섯 건 다 골든이 정확히 나온다.

🔴 **근거 칸(`MAX_EVIDENCE`)을 안 먹는다.** 이것이 `A16`(칸 확대)·`A17`(블록 확대)이
기각된 사유를 피하는 자리다. 실리는 것은 블록이 아니라 **한 줄**이다.

🔴 **유일하게 결정되지 않으면 아무것도 안 낸다.** 틀린 집계를 확신 있게 박아 주는 것이
지금보다 나쁘다 — 모델은 그 줄을 의심하지 않는다. 침묵이 기본값이다.
"""
from __future__ import annotations

import re

#: 질문이 요구하는 연산. 순서가 중요하다 — `증가/감소` 를 먼저 봐야 "얼마나 증가했는가"
#: 가 '합계' 로 안 샌다 (`_EVENT_SUBTYPE` 이 `해지` 를 먼저 두는 것과 같은 이유).
_OPS = (("차", ("얼마나 증가", "얼마나 늘었", "얼마나 감소", "얼마나 줄었",
                "증가했는가", "감소했는가", "차이는")),
        ("몫", ("1척당", "1기당", "1대당", "척당", "기당", "대당", "1건당", "건당",
                "평균 투자금액", "평균 계약금액")),
        ("합", ("합계", "총합", "총액", "모두 더하", "다 더하", "합은")))

#: `몫` 의 분모. 질문에서 읽는다 — 근거에는 척수가 라벨로 안 실린다.
_UNIT_COUNT = re.compile(r"(\d{1,4})\s*(?:척|기|대|건|개)")

#: 연도 필터. `2025년에 공시된` 처럼 **접수일**을 한정하는 표현에서만 읽는다.
_FILED_YEAR = re.compile(r"(20\d{2})\s*년에\s*공시")


def _op(question: str) -> str:
    text = question or ""
    for name, hints in _OPS:
        if any(h in text for h in hints):
            return name
    return ""


def _amount(record: dict):
    value = (record or {}).get("amount")
    return value if isinstance(value, int) and value > 0 else None


def select(records, question: str, route) -> tuple[list[dict], str]:
    """집계 대상과 **필터를 밝힌 설명**. 유일하게 안 정해지면 `([], "")`.

    필터는 두 축뿐이다 — 접수일 연도와 계약상대. R0 이 오답 6건에서 실제로 필요한
    것이 이 둘임을 실측했다(`Q-115` 연도 · `Q-107` 계약상대). 축을 늘리면 유일성이
    깨지고, 유일성이 깨지면 이 모듈은 침묵해야 한다.
    """
    picked = [r for r in (records or []) if _amount(r) is not None]
    if not picked:
        return [], ""
    notes = []

    year = _FILED_YEAR.search(question or "")
    if year:
        want = year.group(1)
        picked = [r for r in picked if str(r.get("rcept_dt", ""))[:4] == want]
        notes.append(f"{want}년 공시분")

    party = getattr(route, "counterparty", None)
    if party:
        from argus.corpus.counterparty import canonical

        key = canonical(party)
        narrowed = [r for r in picked
                    if canonical(r.get("counterparty") or "") == key]
        if narrowed:
            picked, _ = narrowed, notes.append(f"계약상대 {party}")

    return picked, " · ".join(notes)


def line(records, question: str, route) -> str:
    """근거에 실을 한 줄. 낼 것이 없으면 빈 문자열."""
    op = _op(question)
    if not op:
        return ""
    picked, note = select(records, question, route)
    if not picked:
        return ""
    scope = f" ({note})" if note else ""

    if op == "합":
        if len(picked) < 2:
            return ""
        total = sum(_amount(r) for r in picked)
        terms = " + ".join(f"{_amount(r):,}" for r in picked)
        return (f"- [결정적 집계] 위 근거 {len(picked)}건의 금액 합계{scope}: "
                f"{terms} = **{total:,}원**")

    if op == "차":
        if len(picked) != 2:
            return ""
        older, newer = sorted(picked, key=lambda r: str(r.get("rcept_dt", "")))
        delta = _amount(newer) - _amount(older)
        return (f"- [결정적 집계] 금액 변동{scope}: {_amount(newer):,} − "
                f"{_amount(older):,} = **{delta:,}원**")

    if op == "몫":
        if len(picked) != 1:
            return ""
        hit = _UNIT_COUNT.search(question or "")
        if not hit:
            return ""
        n = int(hit.group(1))
        if n <= 1:
            return ""
        total = _amount(picked[0])
        return (f"- [결정적 집계] 단위당 금액{scope}: {total:,} ÷ {n} = "
                f"**{total // n:,}원**")

    return ""
```

def revision_series(records, question: str) -> str:
    """정정 체인의 금액을 **접수일 순서로** 한 줄에 편다. LLM 0회.

    🔴 `Q-023` 의 R0 실측(2026-08-16): 정정본 다섯 건의 금액이 근거에 **전부 있는데**

    """
    if not any(h in (question or "") for h in _SERIES_ASK):
        return ""
    rows = [r for r in (records or []) if _amount(r) is not None]
    if len(rows) < 2:
        return ""
    rows.sort(key=lambda r: (str(r.get("rcept_dt", "")), r.get("rcept_no", "")))
    steps = " → ".join(
        f"{str(r.get('rcept_dt', ''))[:4]}-{str(r.get('rcept_dt', ''))[4:6]}-"
        f"{str(r.get('rcept_dt', ''))[6:8]} {_amount(r):,}원"
        for r in rows)
    return f"- [결정적 집계] 접수일 순 금액 추이 ({len(rows)}건): {steps}"
