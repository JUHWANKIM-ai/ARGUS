```python
@dataclass
class OpenVerdict:
    """Open-question scoring result. The three criteria are not summed.

    `covered` holds one verdict per golden key point. `covers_key_points` is a
    strict majority over that array rather than one judgement over all points
    together: the single boolean let an answer covering three of six points
    score as correct. See docs/open-issues.md item 8.
    """
    covered: tuple[bool, ...]
    grounded: bool
    no_fabrication: bool
    reason: str

    @property
    def covers_key_points(self) -> bool:
        """Strict majority. Five points need three, six need four.

        A tie fails. An empty array fails: an open question with no golden
        points cannot be scored, and reporting it as covered would score the
        absence of a standard as a pass.
        """
        if not self.covered:
            return False
        return sum(self.covered) * 2 > len(self.covered)

    @property
    def correct(self) -> bool:
        return self.covers_key_points and self.grounded and self.no_fabrication
```

```python
def judge_open(question: str, evidence: str, key_points: list[str] | None,
               answer: str, budget: Budget | None = None) -> OpenVerdict:
    """Score an open-ended answer. One LLM call; closed questions are scored by
    normalised exact match and never reach here.

    A parse failure is a scoring failure rather than a wrong answer: `covered`
    is emptied, both remaining axes are set false, and the reason records what
    happened. A `covered` array whose length differs from the point count is
    the same kind of failure — the judge did not count the points — and
    padding it would turn that into a wrong answer.
    """
    points = list(key_points or [])
    result = chat(build_open_judge_messages(question, evidence, key_points, answer),
                  max_tokens=300, temperature=0.0, tag="judge_open")
    if budget:
        budget.spend_llm(result.prompt_tokens, result.completion_tokens)

    match = _JSON.search(result.text or "")
    if not match:
        return OpenVerdict((), False, False, "judge 응답 파싱 실패")
    try:
        data = json.loads(match.group(0))
    except json.JSONDecodeError:
        return OpenVerdict((), False, False, "judge 응답 파싱 실패")

    covered = data.get("covered")
    if not isinstance(covered, list) or len(covered) != len(points):
        got = len(covered) if isinstance(covered, list) else "없음"
        return OpenVerdict((), False, False,
                           f"judge 요점 판정 길이 불일치 (받은 것 {got} · 요점 {len(points)})")

    return OpenVerdict(
        covered=tuple(bool(c) for c in covered),
        grounded=bool(data.get("grounded")),
        no_fabrication=bool(data.get("no_fabrication")),
        reason=str(data.get("reason", "")),
    )
```

```python
SYSTEM_OPEN_JUDGE = """당신은 공시 분석 서술형 답변을 채점하는 심사자입니다.

[모범답안 요소]에는 번호가 매겨져 있습니다. **요소 하나하나마다** 0 또는 1 로

- covered[i]: i번 요소의 내용이 제출답변에 담겼는가
```

```python
def build_open_judge_messages(question: str, evidence: str,
                              key_points: list[str] | None, answer: str) -> list[dict]:
    """Built to the fixed scoring specification: a judgment that drifts makes
    the measurement meaningless.

    The points are numbered because the judge returns one verdict per point in
    that order. An unnumbered list gives the array no anchor.
    """
    points = "\n".join(f"{i}. {p}" for i, p in enumerate(key_points or [], 1)) \
        or "- (없음)"
    user = (f"[근거]\n{(evidence or '')[:MAX_JUDGE_EVIDENCE_CHARS]}\n\n"
            f"[질문]\n{question}\n\n[모범답안 요소]\n{points}\n\n[제출답변]\n{answer}")
    return [{"role": "system", "content": SYSTEM_OPEN_JUDGE},
            {"role": "user", "content": user}]
```

```python
#: Firing words for the holding-ratio series. A separate tuple from
#: `_SERIES_ASK`: widening that one to catch Q-106's "어떻게 변해 왔는지" would
#: move `revision_series`'s firing range too, and that function's behaviour is
#: already measured. This tuple can stay loose because the record condition
#: below — two or more holding filings by one reporter — is the strong half of
#: the premise.
#: See docs/design-rationale.md#holding-series
_HOLDING_ASK = ("추이", "변해", "변화", "변동", "바뀌", "시간순", "시계열", "어떻게 변")


def _ratio_of(record: dict):
    """This filing's holding ratio as a float, or None."""
    value = record.get("ratio_current")
    try:
        return float(value)
    except (TypeError, ValueError):
        return None


def holding_series(records, question: str) -> str:
    """Lay out one reporter's holding ratios in filing order, as one line.

    No LLM call and no retrieval. Returns "" when the question does not ask
    for a history or the records do not carry a series.

    The same move `revision_series` makes for a revision chain's amounts, with
    two differences: the grouping axis is the reporter rather than the chain,
    and the value is `ratio_current` rather than `amount`. The reporter filter
    is load-bearing — the same window holds other reporters' filings on the
    same issuer, and splicing them produces a series that never happened.

    Only the reporter of the top-ranked record is laid out, the same rule
    `revision_series` uses for chains.
    See docs/design-rationale.md#holding-series
    """
    if not any(h in (question or "") for h in _HOLDING_ASK):
        return ""
    rows = [r for r in (records or []) if _ratio_of(r) is not None]
    if len(rows) < 2:
        return ""
    reporter = str(rows[0].get("reporter") or "")
    chain = [r for r in rows if str(r.get("reporter") or "") == reporter]
    if len(chain) < 2:
        return ""
    chain.sort(key=lambda r: (str(r.get("rcept_dt", "")), r.get("rcept_no", "")))

    steps = []
    for index, record in enumerate(chain):
        shares = str(record.get("shares_current") or "")
        # Share counts only at the ends. Golden points 1 and 5 are the first
        # and last counts; carrying all of them doubles the line length for
        # values the question did not ask about.
        tail = f" ({shares}주)" if shares and index in (0, len(chain) - 1) else ""
        steps.append(f"{_ymd(record)} {_ratio_of(record):.2f}%{tail}")

    moved = _ratio_of(chain[0]) - _ratio_of(chain[-1])
    return (f"- [결정적 집계] {reporter} 보고 보유비율 추이 ({len(chain)}건): "
            f"{' → '.join(steps)} · 누적 {-moved:+.2f}%p "
            f"(이 값은 시스템이 위 근거의 비율 필드에서 직접 읽은 것이다. "
            f"답변의 해당 수치는 이 값을 그대로 쓴다.)")
```

```python
#: Marker recording that a project filter applied. Read by `_sum_rows` the
#: same way `_DATE_NOTE` is: both pin the set, where the counterparty axis on
#: its own does not.
_PROJECT_NOTE = "사업분"

#: Separators inside a supply-region field: `체코 남부 Dukovany Site`.
_REGION_SPLIT = re.compile(r"[\s,·/]+")

#: Shortest usable project token. One character matches far too much; two is
#: the length of `체코`, the token Q-107 turns on.
MIN_PROJECT_TOKEN = 2


def _project_token(picked: list[dict], question: str) -> str:
    """The project the question names, read from the evidence, or "".

    The vocabulary is not guessed. A token counts only when it appears in a
    `supply_region` the evidence actually carries AND in the question, so the
    corpus supplies the terms and the question selects among them. This is why
    the axis needs the rebuilt index: before it, no record carries the field
    and this function has nothing to read.

    The longest hit wins, because a more specific place name is a tighter
    filter than the country containing it.
    See docs/design-rationale.md#project-scope
    """
    tokens: set[str] = set()
    for record in picked or []:
        for token in _REGION_SPLIT.split(str(record.get("supply_region") or "")):
            if len(token) >= MIN_PROJECT_TOKEN:
                tokens.add(token)
    hits = [t for t in tokens if t in (question or "")]
    return max(hits, key=len) if hits else ""
```

```python
#: The decisive-line marker, written by every function in this module that
#: places one. Single definition: the repair below finds lines by it, and a
#: second copy would drift from the writers.
DECISIVE_MARK = "- [결정적 집계]"

#: Figures a decisive line has to carry before the repair considers it. Below
#: this the line is a single value, which `substitute` and `state_total`
#: already place without appending a sentence.
GAP_REPAIR_MIN_FIGURES = 3

#: Comma-grouped integers and percentages, the two figure shapes the decisive
#: lines carry.
_LINE_FIGURE = re.compile(r"\d{1,3}(?:,\d{3})+|\d+\.\d+\s*%")

#: Trailing directives inside a decisive line. They instruct the model about
#: the evidence; carried into an answer they are a leak of our own prompt.
_LINE_DIRECTIVE = re.compile(
    r"\((?:이 값은 시스템이|이는 시스템이)[^)]*\)|^\s+답변은 .*$", re.MULTILINE)


def decisive_lines(evidence: str) -> list[str]:
    """The decisive lines placed at the head of the evidence, cleaned.

    The marker and the model-facing directives are removed, leaving the
    computed content alone.
    """
    out = []
    for block in (evidence or "").split("\n\n"):
        if not block.lstrip().startswith(DECISIVE_MARK):
            continue
        text = _LINE_DIRECTIVE.sub("", block)
        text = text.replace(DECISIVE_MARK, "").strip()
        text = re.sub(r"\s*\n\s*", " ", text)
        text = re.sub(r"\s{2,}", " ", text).strip(" ·")
        if text:
            out.append(text)
    return out


def repair_gap(answer: str, evidence: str) -> tuple[str, str]:
    """Append a decisive line's content when the answer dropped most of it.

    No LLM call. Returns (new answer, reason), or (original, "") when nothing
    applies.

    `substitute` and `state_total` reach only questions with a pinned
    operation, because both read `_total`. Q-106 and Q-117 have none, so the
    lines they depend on had no answer-side path at all and the model had to
    copy them. It did not, twice, in measured runs.

    The trigger is the line's own figures: fewer than half of them present in
    the answer. Reading only what we ourselves wrote bounds the false-alarm
    rate structurally, and it catches the case `amount_gap` cannot, where the
    answer states one figure and drops the rest.
    See docs/design-rationale.md#gap-repair
    """
    body, tail = split_tail(answer or "")
    if not body:
        return answer, ""
    additions = []
    for line in decisive_lines(evidence):
        figures = set(_LINE_FIGURE.findall(line))
        if len(figures) < GAP_REPAIR_MIN_FIGURES:
            continue
        present = {f for f in figures if f.replace(" ", "") in body.replace(" ", "")}
        if len(present) * 2 >= len(figures):
            continue
        additions.append(line)
    if not additions:
        return answer, ""
    stated = "\n\n" + "\n".join(f"정리하면 {a}" for a in additions)
    return body + stated + tail, f"repaired {len(additions)}"
```

```python
#: Names of paths outside the corpus we were given. Open issue 1 measured
#: these four shapes; listing them in the system prompt made the model produce
#: them more often (A27, 1 case -> 3), so the list lives here in post
#: processing where it cannot reach the model.
#: See docs/design-rationale.md#external-referral
_EXTERNAL_SOURCE = (
    "네이버", "구글", "증권사", "전자공시시스템", "DART", "다트",
    "공식 웹사이트", "공식 홈페이지", "공식 발표 채널", "관련 기관",
    "홈페이지", "언론", "뉴스",
)

#: Recommending verbs. Both halves have to be in the same sentence: the
#: refusal sentences carry a source name at most, never a recommendation.
_REFERRAL_VERB = (
    "추천", "권장", "권해", "권합니다", "참고하", "참조하",
    "확인해 보", "확인하실", "문의해 보", "알아보시", "바랍니다", "도움이 될",
)

#: Sentence boundary. A period followed by whitespace, or a line break. The
#: separator is captured so the surviving text can be rebuilt with its
#: original spacing: a refusal's two sentences share one paragraph, and
#: rejoining survivors with a fixed separator would split it in two.
_SENTENCE = re.compile(r"((?<=[.!?])\s+|\n+)")

#: A citation the pipeline attached. Never removed: our own citation may name
#: DART, and cutting it removes the evidence.
_HAS_RECEIPT = re.compile(r"\d{14}")


def strip_external_referral(answer: str) -> tuple[str, list[str]]:
    """Remove sentences that recommend a path outside the corpus.

    Returns (new answer, removed sentences). No LLM call.

    A sentence goes only when it names an external source AND recommends
    acting on it. Open issue 1 held that the referral could not be separated
    from the refusal it trails, but the refusal sentences satisfy neither half
    of that conjunction.

    Two sentences are never removed: one carrying a receipt number, because
    that is our own citation, and every sentence, because an emptied answer is
    worse than the defect it fixes.
    See docs/design-rationale.md#external-referral
    """
    if not answer:
        return answer, []
    # Alternating sentence / separator, because the pattern captures. Keeping
    # the separators is what lets an untouched paragraph stay one paragraph.
    parts = _SENTENCE.split(answer)
    kept, removed = [], []
    for index in range(0, len(parts), 2):
        sentence = parts[index]
        separator = parts[index + 1] if index + 1 < len(parts) else ""
        names_source = any(s in sentence for s in _EXTERNAL_SOURCE)
        recommends = any(v in sentence for v in _REFERRAL_VERB)
        if (sentence.strip() and names_source and recommends
                and not _HAS_RECEIPT.search(sentence)):
            removed.append(sentence)
        else:
            kept.append(sentence + separator)
    if not removed:
        return answer, []
    rebuilt = "".join(kept).rstrip()
    if not rebuilt.strip():
        return answer, []
    return rebuilt, removed
```

```python
"""Re-score A43's stored responses under the strict judge.

Usage: python scripts/v8_rejudge.py [--round N] [cohort ...]

The judge changed, so every v1-v7 number was produced by a different scorer
and none of them compares to a v8 number. This runner rebuilds the baseline
from the answers we already have: it reads the stored responses and calls the
judge over them, spending one LLM call per open question and regenerating
nothing. Regenerating would draw new samples and the comparison would then
measure the generator and the scorer at once.

Closed questions are skipped. They are scored by normalised exact match with
no LLM call, and the judge change does not touch them.

Source:  artifacts/ablation/v6/server_<cohort>[_r<n>]/A42/responses/
Output:  artifacts/ablation/v8/rejudge_<cohort>_r<n>/A43/
"""
import json
import sys
import time
from pathlib import Path

sys.path.insert(0, "src")
sys.path.insert(0, ".")
from scripts import error_analysis as ea  # noqa: E402

COHORTS = ("full", "safety", "valuechain")


def source_dir(cohort: str, round_no: int) -> Path:
    name = cohort if round_no == 1 else f"{cohort}_r{round_no}"
    return Path(f"artifacts/ablation/v6/server_{name}/A42/responses")


def out_dir(cohort: str, round_no: int) -> Path:
    return Path(f"artifacts/ablation/v8/rejudge_{cohort}_r{round_no}/A43")


def stored_response(responses: Path, qid: str) -> dict:
    """The stored response body, or FileNotFoundError.

    A missing file is raised rather than skipped: a silently short cohort
    would report a baseline over a different denominator than the arm it is
    compared against.
    """
    path = responses / f"{qid}.json"
    if not path.exists():
        raise FileNotFoundError(f"저장된 응답이 없습니다: {path}")
    return json.loads(path.read_text(encoding="utf-8"))["response"]


def open_cases(cases: list[dict]) -> list[dict]:
    return [c for c in cases if c.get("answer_mode") == "open"]


def rejudge(cohort: str, round_no: int) -> tuple[int, int]:
    responses = source_dir(cohort, round_no)
    target = out_dir(cohort, round_no)
    ea.RESPONSE_DIR = responses
    ea.JUDGE_DIR = target / "judge"
    ea.OUT_DIR = target
    ea.JUDGE_DIR.mkdir(parents=True, exist_ok=True)
    cases = ea.load_cohort(cohort)
    judged = 0
    for index, case in enumerate(open_cases(cases), 1):
        body = stored_response(responses, case["question_id"])
        ea.with_retry(ea.run_judge, case, body, force=True)
        judged += 1
        print("  %-11s %3d %-8s" % (cohort, index, case["question_id"]), flush=True)
        time.sleep(ea.PACE_S)
    rows = ea.analyse(cases, ea._load_records())
    report = ea.build_report(cases, rows, cohort=cohort)
    report["arm"] = "A43"
    (target / "report.json").write_text(
        json.dumps(report, ensure_ascii=False, indent=1), encoding="utf-8")
    correct = sum(1 for r in rows if r["correct"])
    return len(rows), correct


def main(argv) -> int:
    rest = list(argv)
    round_no = 1
    if "--round" in rest:
        i = rest.index("--round")
        round_no = int(rest[i + 1])
        del rest[i:i + 2]
    total = right = 0
    for cohort in (rest or list(COHORTS)):
        n, ok = rejudge(cohort, round_no)
        total += n
        right += ok
        print("%s r%d: %d/%d" % (cohort, round_no, ok, n), flush=True)
    print("합계 r%d: %d/%d" % (round_no, right, total))
    return 0


if __name__ == "__main__":
    sys.exit(main(sys.argv[1:]))
```

```bash
print('\n뒤집힌 문항 %d개: %s' % (len(flips), ' '.join(sorted(flips))))
print('\n=== 놓친 요점 상위 ===')
for k, n in miss.most_common(15):
    print('  %-14s %d회' % (k, n))
```
